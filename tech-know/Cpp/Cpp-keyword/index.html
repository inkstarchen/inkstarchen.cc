    A a1(10);     // 正确：显式调用

    func(A(50));  // 正确：显式构造
}

    class XYPos{...};
class Shape{
    public:
        Shape();
        virtual ~Shape();
        virtual void render();
        void move(const XYPos&);
        virtual void resize();
    protected:
        XYPos center;
};

class Ellipse : public Shape{
    public:
        Ellipse(float maj, float minr);
        virtual void render(); // will define own 
    protected:
        float major_axis, minor_axis;
};

class Circle : public Ellipse{
    public:
        Circle(float radius) : Ellipse(radius, radius){}
        virtual void rendor();
};

void render(Shape* p) {
    p->render(); // calls correct render function
} // for given Shape!

void func() {
    Ellipse ell(10, 20);
    ell.render(); // static -- Ellipse::render();

    Circle circ(40);
    circ.render(); // static -- Circle::render();

    render(&ell); // dynamic -- Ellipse::render();
    render(&circ); // dynamic -- Circle::render()
}
